*&  Form compute_abc
*&---------------------------------------------------------------------*
FORM compute_abc.
*
* Sort per valore
  SORT tab_lay_cd BY wertn DESCENDING.
*
* 1 - Materiali
* Definizione valore limite a, b, c
  CLEAR: value_a, value_b, value_c.
  value_a = value_m * 70 / 100.
  value_b = value_a + ( value_m * 20 / 100 ).
  value_c = value_b + ( value_m * 10 / 100 ).
  CLEAR: value_m.
  LOOP AT tab_lay_cd WHERE tipo = 'M'.
    ADD tab_lay_cd-wertn TO value_m.
    IF value_m < value_a.
      tab_lay_cd-class = 'A'.
    ELSEIF value_m > value_a AND value_m < value_b.
      tab_lay_cd-class = 'B'.
    ELSEIF value_m > value_b.
      tab_lay_cd-class = 'C'.
    ENDIF.
    MODIFY tab_lay_cd.
  ENDLOOP.
  PERFORM stampa_report_alv_cdo.
*
ENDFORM.                    " compute_abc
